    const playTone = (type) => {
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator(); 
        const gainNode = ctx.createGain();
        osc.connect(gainNode); 
        gainNode.connect(ctx.destination);
        
        if (type === 'great') { 
          osc.type = 'sine'; 
          osc.frequency.setValueAtTime(1046.50, ctx.currentTime); 
          osc.frequency.exponentialRampToValueAtTime(2093.00, ctx.currentTime + 0.1); 
          gainNode.gain.setValueAtTime(0.15, ctx.currentTime); 
          gainNode.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.4); 
          osc.start(); 
          osc.stop(ctx.currentTime + 0.4); 
        } else if (type === 'success') {

    const generateLevel = (score) => {
      const difficulty = Math.floor(score / 300);
      let baseCanvases = 1;
      
      if (difficulty >= 2) baseCanvases = Math.random() > 0.6 ? 2 : 1; 
      if (difficulty >= 4) baseCanvases = Math.random() > 0.5 ? 3 : 2; 
      if (difficulty >= 6) baseCanvases = 3;
      
      const numCanvases = Math.max(GAME_CONFIG.sequentialCanvases.min, Math.min(baseCanvases, GAME_CONFIG.sequentialCanvases.max));
      const canvases = [];

      for (let c = 0; c < numCanvases; c++) {
        let baseBars = 1;
        if (difficulty >= 1) baseBars = Math.random() > 0.5 ? 2 : 1; 
        if (difficulty >= 3) baseBars = Math.random() > 0.6 ? 3 : 2; 
        if (difficulty >= 5) baseBars = 3;
        
        if (numCanvases === 3 && baseBars > 2) baseBars = 2;
        
        const numBars = Math.max(GAME_CONFIG.concentricRings.min, Math.min(baseBars, GAME_CONFIG.concentricRings.max));
        const bars = [];
        
        let calculatedSpeed = GAME_CONFIG.speed.min + (difficulty * 0.3) + (Math.random() * 1.0);
        let baseSharedSpeed = Math.max(GAME_CONFIG.speed.min, Math.min(calculatedSpeed, GAME_CONFIG.speed.max));
        let directionMultiplier = (difficulty >= 3 && c % 2 !== 0) ? -1 : 1; 
        const sharedSpeed = baseSharedSpeed * directionMultiplier;
        const winningDistance = Math.abs(sharedSpeed) * (90 + (Math.random() * 60));
        
        const getRadius = (index) => { 
          const baseRadii = [80, 55, 30]; 
          return index < baseRadii.length ? baseRadii[index] : Math.max(10, 30 - ((index - 2) * 15)); 
        };
        
        let calculatedTargetWidth = 70 - (difficulty * 5) + (Math.random() * 20);
        const targetWidth = Math.max(GAME_CONFIG.targetSize.min, Math.min(GAME_CONFIG.targetSize.max, calculatedTargetWidth)); 
        let calculatedGreatWidth = targetWidth * 0.25;
        const greatWidth = Math.max(GAME_CONFIG.greatSize.min, Math.min(GAME_CONFIG.greatSize.max, calculatedGreatWidth));
        
        const targetStart = Math.random() * 360; 
        let initialPos = ((targetStart + (greatWidth / 2)) - (winningDistance * directionMultiplier)) % 360;
        if (initialPos < 0) initialPos += 360;

        for (let i = 0; i < numBars; i++) {
          const ringOffset = i * 75;
          bars.push({ 
            id: i, 
            radius: getRadius(i), 
            speed: sharedSpeed, 
            targetStart: (targetStart + ringOffset) % 360, 
            targetWidth: targetWidth, 
            greatWidth: greatWidth, 
            initialPos: (initialPos + ringOffset) % 360 
          });
        }
        canvases.push(bars);
      }
      return canvases;
    };

const processQueue = async () => {
console.log("⏳ 1. processQueue invocado. Procesando:", isProcessingQueueRef.current, "| Cola:", scoreQueueRef.current.length);

  
  if (isProcessingQueueRef.current) return;

  
  if (scoreQueueRef.current.length === 0) {
    console.log("✅ 2. Cola vacía. Apagando estado de carga de la tabla...");
    setIsLeaderboardLoading(false); 
    return;
  }
  
  isProcessingQueueRef.current = true;
  setIsLeaderboardLoading(true);
  
  const task = scoreQueueRef.current[0];
  let success = false; 
  let attempt = 0;
  
  while (!success && attempt < 5) {
    try {
      console.log(`🚀 3. Intento ${attempt + 1}: Enviando datos al servidor...`);
      const result = await executeSaveOnServer(task.empId, task.score, task.mode, task.duration);
      
      console.log("📥 4. Respuesta recibida del servidor:", result);
      if (result && result.success) {
        setLeaderboard(result.leaderboard);
        if (result.savedEmpId) setLastSavedEmpId(result.savedEmpId);
        success = true;
      } else if (result && !result.success && result.reason === 'LIMIT_REACHED') {
        setLeaderboard(result.leaderboard);
        setToastInfo(`⚠️ Puntuación no registrada: Has superado el límite de intentos.`);
        setTimeout(() => setToastInfo(null), 5000);
        success = true; 
      } else { 
        throw new Error("Respuesta inválida"); 
      }
    } catch (err) {
      console.error("❌ 5. Error detectado en el intento:", err);
      attempt++;
      if (attempt < 5) {
      console.log("🔄 Reintentando en breve...");
        await new Promise(resolve => setTimeout(resolve, Math.pow(2, attempt) * 1000));
      }
    }
  }
  
  
  console.log("🧹 6. Limpiando tarea completada y quitando candado.");
  scoreQueueRef.current.shift();
  isProcessingQueueRef.current = false;
  
  
  processQueue();
};

            google.script.run

                          const formattedLb = (lbData || []).map(item => ({
                            id: item.id || item.employeeId || item.name, 
                            name: item.name,
                            score: item.score
                          }));
                          
                          
                          resolve({

                        .withFailureHandler(reject)
                        .getLeaderboardFromSheet(CURRENT_GAME, "Todos");

              })
              .getInstructions(CURRENT_GAME);
          } else {
            // Entorno de prueba (Local)
            setTimeout(() => {
              setInstructions([
                { title: "Monitor de Signos", desc: "Este dispositivo muestra la frecuencia en tiempo real.", img: "https://placehold.co/400x600/10b981/050505?text=Vertical" },
                { title: "Desfibrilador", desc: "Equipo crítico para restablecer el ritmo cardíaco.", img: "https://placehold.co/400x300/c21f32/ffffff?text=Horizontal" }
              ]);
              setIsLoadingInstructions(false);
            }, 1000);
          }
        }
      };

      const handleNextOrClose = () => {
        if (currentIndex < instructions.length - 1) {
          setCurrentIndex(prev => prev + 1);
        } else {
          setShowModal(false);
        }
      };

      return (
        <div className={`relative z-10 w-full max-w-lg p-6 transition-all duration-700 transform ${isVisible ? 'translate-y-0 opacity-100 scale-100' : 'translate-y-12 opacity-0 scale-95'}`}>
          <div className="relative overflow-hidden rounded-3xl border border-white/20 shadow-xl p-10 flex flex-col items-center text-center bg-black/80">
            
            <div className="rounded-xl overflow-hidden mb-4">
               <img src={SAFE_URLS.LOGO} alt="Plexus Logo" className="h-20 object-contain" />
            </div>
            
            <p className="text-neutral-300 uppercase tracking-[0.2em] text-sm font-medium mb-8">Ajuste Crítico Pro</p>
            
            <div className="relative mb-6">
              <img 
                src={"https://drive.google.com/thumbnail?id=19FTVhVxfRAB0r-7p8pdcmWHZMqRVzCSl&sz=w400"} 
                style={{ width: "350px", animation: "float 3s ease-in-out infinite" }} 
                alt="Paciente"
              />  
            </div>
            
            {GAME_CONFIG.isGameActive ? (
              <button 
                onPointerDown={(e) => { e.preventDefault(); if (isVisible) onStart(); }} 
                className="w-full relative group overflow-hidden rounded-xl border border-plexus-400/50 py-5 mb-3 shadow-lg hover:scale-[1.03] active:scale-95 transition-all duration-300 bg-plexus-600/80 hover:bg-plexus-500/90"
              >
                <div className="relative flex items-center justify-center gap-3">
                  <IconZap className="w-6 h-6 text-white" />
                  <span className="text-white font-bold tracking-wider text-lg uppercase">Iniciar Secuencia</span>
                </div>
              </button>
            ) : (
              <div className={`w-full rounded-xl border py-5 mb-3 shadow-inner flex flex-col items-center justify-center gap-1 ${GAME_CONFIG.campaignError ? 'border-red-500/50 bg-red-950/40 opacity-100' : 'border-neutral-700 bg-black/50 opacity-70'}`}>
                <IconLock className={`w-6 h-6 mb-1 ${GAME_CONFIG.campaignError ? 'text-red-400' : 'text-neutral-500'}`} />
                <span className={`font-bold tracking-wider text-sm uppercase ${GAME_CONFIG.campaignError ? 'text-red-400' : 'text-neutral-400'}`}>
                  {GAME_CONFIG.campaignError ? 'Error en Base de Datos' : 'Campaña Inactiva'}
                </span>
                <span className={`text-[10px] uppercase font-mono tracking-widest text-center px-4 ${GAME_CONFIG.campaignError ? 'text-red-300' : 'text-neutral-600'}`}>
                  {GAME_CONFIG.campaignError || GAME_CONFIG.campaignName}
                </span>
              </div>
            )}
            
            <button 
              onPointerDown={(e) => { e.preventDefault(); if (isVisible) onLeaderboard(); }} 
              className="w-full rounded-xl border border-white/20 py-4 flex items-center justify-center gap-3 hover:scale-[1.03] active:scale-95 transition-all duration-300 bg-black/60 hover:bg-black/40"
            >
              <IconTrophy className="w-5 h-5 text-yellow-400" />
              <span className="text-neutral-200 font-bold tracking-wider text-sm uppercase">Mejores Puntajes</span>
            </button>

            {/* Botón de Instrucciones: Más estrecho (w-4/5), color blanco opaco (neutral-200/90) y letras oscuras */}
            <button 
              onPointerDown={handleOpenInstructions} 
              className="w-4/5 mx-auto mt-4 rounded-lg border border-neutral-300/50 py-2.5 flex items-center justify-center gap-2 hover:scale-[1.02] active:scale-95 transition-all duration-300 bg-neutral-200/90 hover:bg-neutral-100"
            >
              <IconInfo className="w-4 h-4 text-neutral-800" />
              <span className="text-neutral-900 font-bold tracking-widest text-xs uppercase">Instrucciones</span>
            </button>
            
          </div>

          {/* --- MODAL DE INSTRUCCIONES --- */}
          {showModal && (
            <div className="absolute inset-0 z-50 flex items-center justify-center p-4 bg-black/90 backdrop-blur-sm transition-opacity duration-300">
              <div className="relative w-full max-w-sm rounded-3xl border-2 border-red-600/60 overflow-hidden flex flex-col items-center p-6 bg-black shadow-2xl animate-zoom-in">
                
                {/* Título y un margen separador (border-bottom) de color rojo */}
                <h2 className="text-xl font-black text-red-500 uppercase tracking-widest mb-4 border-b border-red-500/50 pb-3 w-full text-center">
                  Instrucciones
                </h2>

                {isLoadingInstructions ? (
                  <div className="flex flex-col items-center justify-center py-10">
                    <div className="w-10 h-10 border-4 border-neutral-300/30 border-t-neutral-300 rounded-full animate-spin mb-4"></div>
                    <span className="text-neutral-400 font-mono text-xs uppercase animate-pulse">Sincronizando...</span>
                  </div>
                ) : (
                  <div className="w-full flex flex-col items-center transition-all duration-500">
                    
                    <div className="flex gap-2 mb-4">
                      {instructions.map((_, idx) => (
                        <div key={idx} className={`h-1.5 rounded-full transition-all duration-300 ${idx === currentIndex ? 'w-6 bg-neutral-300' : 'w-2 bg-neutral-700'}`} />
                      ))}
                    </div>

                    {/* Margen blanco opaco para la tarjeta interna */}
                    <div className="w-full bg-neutral-900/50 rounded-xl border border-neutral-200/30 p-3 mb-6 flex flex-col gap-3">
                      
                      {instructions[currentIndex]?.img && (
                        <div className="w-full h-auto min-h-[150px] max-h-[50vh] bg-black rounded-lg overflow-hidden flex items-center justify-center border border-neutral-800">
                          <img 
                            src={instructions[currentIndex].img} 
                            alt="Instrucción" 
                            className="w-full h-full object-contain"
                          />
                        </div>
                      )}
                      
                      <div className="flex flex-col text-left mt-2">
                        <span className="font-black text-lg text-red-500 uppercase tracking-wider mb-2">
                          {instructions[currentIndex]?.title}
                        </span>
                        <span className="text-neutral-300 text-sm leading-relaxed overflow-y-auto custom-scrollbar max-h-32">
                          {instructions[currentIndex]?.desc}
                        </span>
                      </div>
                    </div>

                    {/* Botón dinámico: Blanco opaco para "Siguiente", Rojo para "Cerrar" */}
                    <button 
                      onPointerDown={(e) => { e.preventDefault(); handleNextOrClose(); }}
                      className={`w-full py-4 rounded-xl font-bold text-lg uppercase transition-all duration-300 active:scale-95 border ${
                        currentIndex < instructions.length - 1 
                          ? 'bg-neutral-200/90 hover:bg-neutral-100 border-neutral-300/50 text-neutral-900' 
                          : 'bg-red-600/80 hover:bg-red-500/90 border-red-500/50 text-white'
                      }`}
                    >
                      {currentIndex < instructions.length - 1 ? 'Siguiente' : 'Cerrar'}
                    </button>
                  </div>
                )}
                
              </div>
            </div>
          )}
          {/* -------------------------------------- */}
          
        </div>
      );
    }

    /* =========================================================================
       PANTALLA DE JUEGO (GAMEPLAY)
       ========================================================================= */
    function GameScreen({ onGameOver, onQuit, setGlobalInvincible, isVisible }) {
      const [score, setScore] = useState(0); 
      const [lives, setLives] = useState(GAME_CONFIG.initialLives); 
      const [isPaused, setIsPaused] = useState(false); 
      const [isExiting, setIsExiting] = useState(false); 
      const [isPauseMounted, setIsPauseMounted] = useState(false); 
      const [showPauseMenu, setShowPauseMenu] = useState(false); 
      const [feedback, setFeedback] = useState(null); 
      const [feedbackVisible, setFeedbackVisible] = useState(false); 
      const [showContinue, setShowContinue] = useState(false); 
      const [levelConfig, setLevelConfig] = useState([]); 
      const [activeCanvasIndex, setActiveCanvasIndex] = useState(0); 
      const [normalStreak, setNormalStreak] = useState(0); 
      const [perfectStreak, setPerfectStreak] = useState(0); 
      const [invincibleCanvasesLeft, setInvincibleCanvasesLeft] = useState(0); 
      
      const isInvincible = invincibleCanvasesLeft > 0; 
      
      const [visualNormalMax, setVisualNormalMax] = useState(false); 
      const [visualPerfectMax, setVisualPerfectMax] = useState(false);
      
      const positionsRef = useRef([]); 
      const requestRef = useRef(); 
      const indicatorsRef = useRef([]); 
      const inputLockedRef = useRef(false); 
      
      const [isRadarEntering, setIsRadarEntering] = useState(false); 
      const gameStartTimeRef = useRef(Date.now()); 
      const [triviaOfTheDay, setTriviaOfTheDay] = useState(null); 
      
      const stateRef = useRef({ isPaused, isExiting, feedback, levelConfig, showContinue, lives, score, perfectStreak, normalStreak, invincibleCanvasesLeft, activeCanvasIndex });

      useEffect(() => { 
        stateRef.current = { isPaused, isExiting, feedback, levelConfig, showContinue, lives, score, perfectStreak, normalStreak, invincibleCanvasesLeft, activeCanvasIndex }; 
      }, [isPaused, isExiting, feedback, levelConfig, showContinue, lives, score, perfectStreak, normalStreak, invincibleCanvasesLeft, activeCanvasIndex]);

      const initLevel = useCallback((currentScore, isNewGame) => {
        const canvases = generateLevel(currentScore); 
        setIsRadarEntering(true); 
        setLevelConfig(canvases); 
        setActiveCanvasIndex(0);
        positionsRef.current = canvases.map(canvasBars => canvasBars.map(bar => ({ pos: bar.initialPos }))); 
        indicatorsRef.current = []; 
        setFeedback(null); 
        setFeedbackVisible(false); 
        setShowContinue(false); 
        setVisualNormalMax(false); 
        setVisualPerfectMax(false); 
        setIsExiting(false); 
        inputLockedRef.current = false;
        
        if (isNewGame) { 
          setNormalStreak(0); 
          setPerfectStreak(0); 
          setInvincibleCanvasesLeft(0); 
          setGlobalInvincible(false); 
          setLives(GAME_CONFIG.initialLives); 
          gameStartTimeRef.current = Date.now(); 
        }
      }, [setGlobalInvincible]);

      useEffect(() => { initLevel(0, true); }, [initLevel]);

      useEffect(() => { 
        if (isRadarEntering) { 
          const timer = setTimeout(() => setIsRadarEntering(false), 1000); 
          return () => clearTimeout(timer); 
        } 
      }, [isRadarEntering]);

      const triggerPause = () => { 
        if (!isVisible) return; 
        setIsPaused(true); 
        setIsPauseMounted(true); 
        setTimeout(() => setShowPauseMenu(true), 10); 
      };

      const handleResume = () => { 
        setIsPaused(false); 
        setShowPauseMenu(false); 
        setTimeout(() => setIsPauseMounted(false), 500); 
      };

      const handleQuitFromPause = () => { 
        setShowPauseMenu(false); 
        setIsExiting(true); 
        setTimeout(() => { 
          setIsPaused(false); 
          setIsPauseMounted(false); 
          onQuit(); 
        }, 400); 
      };

      const animate = useCallback(() => {
        requestRef.current = requestAnimationFrame(animate);
        const { isPaused, isExiting, feedback, levelConfig, activeCanvasIndex } = stateRef.current;
        
        if (isPaused || isExiting || feedback || levelConfig.length === 0) return;
        
        if (activeCanvasIndex < levelConfig.length) {
            positionsRef.current[activeCanvasIndex].forEach((p, idx) => {
              const bar = levelConfig[activeCanvasIndex][idx]; 
              if (!bar) return;
              p.pos = (p.pos + bar.speed + 360) % 360;
              if (indicatorsRef.current[activeCanvasIndex] && indicatorsRef.current[activeCanvasIndex][idx]) { 
                indicatorsRef.current[activeCanvasIndex][idx].style.transform = `rotate(${p.pos}deg)`; 
              }
            });
        }
      }, []);

      useEffect(() => { 
        requestRef.current = requestAnimationFrame(animate); 
        return () => cancelAnimationFrame(requestRef.current); 
      }, [animate]);

      const processHit = useCallback(() => {
        const { isPaused, isExiting, feedback, levelConfig, showContinue, perfectStreak, normalStreak, lives, invincibleCanvasesLeft, activeCanvasIndex } = stateRef.current;
        if (isPaused || isExiting || feedback || levelConfig.length === 0 || showContinue || activeCanvasIndex >= levelConfig.length) return;
        if (inputLockedRef.current) return; 

        const currentCanvas = levelConfig[activeCanvasIndex]; 
        const bar = currentCanvas[0]; 
        const currentPos = positionsRef.current[activeCanvasIndex][0].pos;
        
        let hit = false; let greatHit = false;
        const endAngle = bar.targetStart + bar.targetWidth; 
        const greatEndAngle = bar.targetStart + bar.greatWidth;

        if (endAngle > 360) { if (currentPos >= bar.targetStart || currentPos <= (endAngle % 360)) hit = true; } 
        else { if (currentPos >= bar.targetStart && currentPos <= endAngle) hit = true; }

        if (greatEndAngle > 360) { if (currentPos >= bar.targetStart || currentPos <= (greatEndAngle % 360)) greatHit = true; } 
        else { if (currentPos >= bar.targetStart && currentPos <= greatEndAngle) greatHit = true; }

        const wasInvincibleActive = invincibleCanvasesLeft > 0; 
        if (wasInvincibleActive) hit = true;

        if (hit) {
          inputLockedRef.current = true; setTimeout(() => inputLockedRef.current = false, 300); 
          const pointsEarned = greatHit ? 200 : 100; setScore(prev => prev + pointsEarned);

          if (activeCanvasIndex < levelConfig.length - 1) { 
            playTone(greatHit ? 'great' : 'success'); setActiveCanvasIndex(prev => prev + 1); 
          } else {
            setFeedback(greatHit ? 'great' : 'success'); playTone(greatHit ? 'great' : 'success'); setTimeout(() => setFeedbackVisible(true), 10); 
            
            if (!wasInvincibleActive) {
              if (greatHit) {
                const nextPerfectStreak = perfectStreak + 1; const nextNormalStreak = normalStreak + 1;
                if (nextPerfectStreak >= 3) { setInvincibleCanvasesLeft(3); setGlobalInvincible(true); setVisualPerfectMax(true); setPerfectStreak(0); setNormalStreak(0); playTone('powerup'); setTimeout(() => setVisualPerfectMax(false), 2000); } 
                else { setPerfectStreak(nextPerfectStreak); if (nextNormalStreak >= 3) { setLives(prev => Math.min(prev + 1, GAME_CONFIG.initialLives)); setVisualNormalMax(true); setNormalStreak(0); playTone('powerup'); setTimeout(() => setVisualNormalMax(false), 2000); } else { setNormalStreak(nextNormalStreak); } }
              } else {
                setPerfectStreak(0); const nextNormalStreak = normalStreak + 1;
                if (nextNormalStreak >= 3) { setLives(prev => Math.min(prev + 1, GAME_CONFIG.initialLives)); setVisualNormalMax(true); setNormalStreak(0); playTone('powerup'); setTimeout(() => setVisualNormalMax(false), 2000); } else { setNormalStreak(nextNormalStreak); }
              }
            }
            if (wasInvincibleActive) { setInvincibleCanvasesLeft(prev => { const next = prev - 1; if (next <= 0) setGlobalInvincible(false); return next; }); }
            setTriviaOfTheDay(TRIVIA_CATALOG[Math.floor(Math.random() * TRIVIA_CATALOG.length)]); setTimeout(() => setShowContinue(true), 600);
          }
        } else {
          inputLockedRef.current = true; setTimeout(() => inputLockedRef.current = false, 300);
          setNormalStreak(0); setPerfectStreak(0); setFeedback('fail'); playTone('fail');
          setLives(prev => { const newLives = prev - 1; if (newLives <= 0) playTone('gameover'); return newLives; });
          setTriviaOfTheDay(TRIVIA_CATALOG[Math.floor(Math.random() * TRIVIA_CATALOG.length)]); setTimeout(() => setFeedbackVisible(true), 10); setTimeout(() => setShowContinue(true), 600);
        }
      }, [setGlobalInvincible]);

      const handleContinue = useCallback(() => {
        if (!isVisible) return; 
        const { lives, score } = stateRef.current; setShowContinue(false); setFeedbackVisible(false); 
        setTimeout(() => { if (lives <= 0) { const durationInSeconds = Math.floor((Date.now() - gameStartTimeRef.current) / 1000); onGameOver(score, durationInSeconds); } else { initLevel(score); } }, 500); 
      }, [onGameOver, initLevel, isVisible]);

      useEffect(() => { const handleKeyDown = (e) => { if (e.code === 'Space') { e.preventDefault(); if (stateRef.current.showContinue) handleContinue(); else processHit(); } }; window.addEventListener('keydown', handleKeyDown); return () => window.removeEventListener('keydown', handleKeyDown); }, [processHit, handleContinue]);

      const radarSizeClass = levelConfig.length === 1 ? 'w-72 h-72' : (levelConfig.length === 2 ? 'w-40 h-40' : 'w-32 h-32'); 

      return (
        <div className={`relative z-10 w-full min-h-screen h-[100dvh] max-w-lg mx-auto flex flex-col justify-between border-x border-white/10 shadow-2xl transition-all duration-700 transform ${isVisible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-12'} bg-black/80`} style={{ backfaceVisibility: 'hidden' }}>
          
          <div className="p-6 flex justify-between items-center border-b border-white/10 relative z-10 shrink-0 transition-all duration-500 bg-black/60">
            <div className="flex flex-col"><span className="text-xs text-emerald-400 font-bold uppercase tracking-widest">Puntuación</span><span className="text-3xl font-mono font-black text-white">{score}</span></div>
            <img src={SAFE_URLS.LOGO} alt="Plexus" className="h-6 rounded object-contain opacity-80" />
            <div className="flex items-center gap-4">
              <div className="flex flex-col items-end gap-1.5 mr-2">
                <div className="flex items-center gap-1.5"><span className="text-[10px] text-emerald-500 font-bold uppercase tracking-wider">Combo</span><div className="flex gap-0.5">{[1, 2, 3].map(step => ( <div key={step} className={`w-2.5 h-1.5 rounded-sm transition-all duration-300 ${visualNormalMax ? 'bg-emerald-400 animate-pulse' : (step <= normalStreak ? 'bg-emerald-500' : 'bg-black/60 border border-white/10')}`} /> ))}</div></div>
                <div className="flex items-center gap-1.5"><span className="text-[10px] text-yellow-500 font-bold uppercase tracking-wider">Perfecto</span><div className="flex gap-0.5">{[1, 2, 3].map(step => ( <div key={step} className={`w-2.5 h-1.5 rounded-sm transition-all duration-300 ${visualPerfectMax ? 'bg-yellow-400 animate-pulse' : (step <= perfectStreak ? 'bg-yellow-500' : 'bg-black/60 border border-white/10')}`} /> ))}</div></div>
              </div>
              <div className="flex gap-1">{Array.from({ length: GAME_CONFIG.initialLives }, (_, i) => i + 1).map(i => ( <IconHeart key={i} filled={i <= lives} className={`w-6 h-6 ${i <= lives ? 'text-red-500' : 'text-neutral-700'}`} /> ))}</div>
              <button onPointerDown={(e) => { e.preventDefault(); triggerPause(); }} className="p-3 bg-white/10 rounded-full active:bg-white/20"><IconPause className="w-6 h-6 text-white" /></button>
            </div>
          </div>

          <div className="flex-1 flex flex-col items-center justify-center relative p-4 gap-4">
            <div className={`flex flex-row flex-wrap justify-center items-center gap-6 w-full ${isExiting ? 'animate-crt-shutdown' : ''}`}>
              {levelConfig.map((canvasBars, cIdx) => {
                const isSolved = cIdx < activeCanvasIndex; const isActive = cIdx === activeCanvasIndex; const isPending = cIdx > activeCanvasIndex;
                let stateBorder = 'border-neutral-700 bg-black/60';
                if (isSolved) stateBorder = 'border-neutral-500 bg-black/80 grayscale opacity-40';
                if (isActive && feedback === 'great') stateBorder = 'border-white bg-white/20';
                if (isActive && feedback === 'success') stateBorder = 'border-emerald-500 bg-emerald-500/10';
                if (isActive && feedback === 'fail') stateBorder = 'border-red-500 bg-red-500/10 animate-shake';
                if (isPending) stateBorder = 'border-neutral-800 bg-black/40 opacity-30 grayscale';
                const animationClass = isRadarEntering ? 'animate-hologram-boot' : '';
                return (
                  <div key={cIdx} className={`relative ${radarSizeClass} rounded-full border-4 ${stateBorder} overflow-hidden transition-all duration-300 flex-shrink-0 ${animationClass}`} style={{ animationDelay: isRadarEntering ? `${cIdx * 150}ms` : '0ms', backfaceVisibility: 'hidden', willChange: 'transform, opacity' }}>
                    <div className="absolute inset-0 flex items-center justify-center opacity-20 pointer-events-none"><svg width="60%" height="60%" viewBox="0 0 100 100" className="text-plexus-400"><rect x="15" y="15" width="70" height="70" rx="8" fill="none" stroke="currentColor" strokeWidth="2" strokeDasharray="4 4" /><circle cx="50" cy="50" r="15" fill="none" stroke="currentColor" strokeWidth="2" /><path d="M50 15 L50 85 M15 50 L85 50" stroke="currentColor" strokeWidth="1" strokeDasharray="2 2" /></svg></div>
                    {/* ARQUITECTURA: Patrón interior del radar restaurado */}
                    {isSolved && <div className="absolute inset-0 z-20 bg-black/40 pattern-bubble opacity-50"></div>}
                    <svg viewBox="0 0 200 200" className="absolute inset-0 w-full h-full text-white z-10">
                      {canvasBars.map((bar, idx) => {
                        const circ = 2 * Math.PI * bar.radius; const targetDash = `${(bar.targetWidth / 360) * circ} ${circ}`; const greatDash = `${(bar.greatWidth / 360) * circ} ${circ}`;
                        let trackStroke = "rgba(255,255,255,0.1)"; let targetStroke = "#34d399"; let greatStroke = "#ffffff"; let needleColor = bar.speed < 0 ? "#8b5cf6" : "#c21f32"; let needleBg = bar.speed < 0 ? "#8b5cf6" : "#c21f32"; let ringOpacity = "opacity-100 transition-opacity duration-300";
                        if (isPending) { trackStroke = "rgba(255,255,255,0.02)"; targetStroke = "#374151"; greatStroke = "#4b5563"; needleColor = "#4b5563"; needleBg = "#374151"; ringOpacity = "opacity-30 transition-opacity duration-300"; } 
                        else if (isSolved) { trackStroke = "rgba(255,255,255,0.04)"; targetStroke = "rgba(52, 211, 153, 0.4)"; greatStroke = "rgba(255, 255, 255, 0.4)"; needleColor = "rgba(16, 185, 129, 0.6)"; needleBg = "rgba(16, 185, 129, 0.4)"; ringOpacity = "opacity-60 transition-opacity duration-300"; }
                        if (isInvincible && !isPending && !isSolved) { needleColor = "#ffffff"; needleBg = "#ffffff"; }
                        return (
                          <g key={bar.id} className={ringOpacity}>
                            <circle cx="100" cy="100" r={bar.radius} fill="none" stroke={trackStroke} strokeWidth="12" />
                            <circle cx="100" cy="100" r={bar.radius} fill="none" stroke={(isActive && feedback === 'fail') ? '#ef4444' : targetStroke} strokeWidth="12" strokeDasharray={targetDash} strokeDashoffset="0" transform={`rotate(${bar.targetStart - 90}, 100, 100)`} />
                            <circle cx="100" cy="100" r={bar.radius} fill="none" stroke={(isActive && feedback === 'fail') ? 'transparent' : greatStroke} strokeWidth="12" strokeDasharray={greatDash} strokeDashoffset="0" transform={`rotate(${bar.targetStart - 90}, 100, 100)`} />
                            <g ref={el => { if (!indicatorsRef.current[cIdx]) indicatorsRef.current[cIdx] = []; indicatorsRef.current[cIdx][idx] = el; }} style={{ transform: `rotate(${bar.initialPos}deg)`, transformOrigin: '100px 100px', willChange: 'transform' }}>
                              <line x1="100" y1={100 - bar.radius + 8} x2="100" y2={100 - bar.radius - 8} stroke={(isActive && feedback === 'fail') ? '#ef4444' : needleColor} strokeWidth="4" strokeLinecap="round" />
                              <circle cx="100" cy={100 - bar.radius} r="2" fill={(isActive && feedback === 'fail') ? '#ef4444' : needleBg} />
                            </g>
                          </g>
                        );
                      })}
                    </svg>
                  </div>
                );
              })}
            </div>
            
            {/* Modal Feedback Dinámico */}
            {feedback && (
              <div className={`absolute inset-0 z-40 flex items-center justify-center pointer-events-none transition-all duration-500 transform ${feedbackVisible ? 'opacity-100 scale-100' : 'opacity-0 scale-90'}`}>

      return (

    /* =========================================================================
       PANTALLA DE LEADERBOARD
       ========================================================================= */
    function LeaderboardRow({ entry, rank, isHighlight, delay }) {
      const [displayScore, setDisplayScore] = useState(entry.score);
      const [animClass, setAnimClass] = useState('animate-row-enter');
      const rowRef = useRef(null);

      useEffect(() => {
        if (entry.score > displayScore) {
          setAnimClass('animate-flip-update');
          setTimeout(() => { setDisplayScore(entry.score); }, 400); 
          setTimeout(() => { setAnimClass(''); }, 800);
        }
      }, [entry.score]);

      useEffect(() => {
        if (isHighlight && rowRef.current) {
          setTimeout(() => { rowRef.current.scrollIntoView({ behavior: 'smooth', block: 'center' }); }, 800);
        }
      }, [isHighlight]);

      const rankColors = [
        'text-yellow-400 border-yellow-400/40 bg-yellow-400/5', 
        'text-neutral-300 border-neutral-300/30 bg-white/5', 
        'text-amber-600 border-amber-600/30 bg-amber-600/5' 
      ];
      
      const isPodium = rank <= 3;
      const highlightedStyle = isHighlight 
        ? 'border-2 border-emerald-400 bg-emerald-500/10' 
        : (isPodium ? `border ${rankColors[rank-1]}` : 'border border-white/5 bg-black/40');

      return (
        <div ref={rowRef} className={`flex justify-between items-center rounded-lg p-4 transition-all duration-500 ${highlightedStyle} ${animClass}`} style={animClass === 'animate-row-enter' ? { animationFillMode: 'both', animationDelay: `${delay}ms` } : {}}>
          <div className="flex items-center gap-4 flex-1 min-w-0">
            <span className={`font-black text-xl w-6 shrink-0 text-center ${isPodium ? '' : 'text-neutral-600'}`}>#{rank}</span>
            <span className={`font-mono font-bold tracking-wider truncate max-w-[180px] sm:max-w-[250px] ${isPodium ? 'text-white text-lg' : 'text-neutral-300 text-base'}`}>{entry.name}</span>
          </div>
          <span className={`font-mono font-black shrink-0 ml-2 ${isPodium ? 'text-2xl text-white' : 'text-xl text-emerald-400'}`}>{displayScore}</span>
        </div>
      );
    }

    function LeaderboardScreen({ scores, isLoading, lastSavedEmpId, onBack, isVisible }) {
      const [searchQuery, setSearchQuery] = useState('');

      const filteredScores = scores.filter(entry => 
        (entry.name || '').toLowerCase().includes(searchQuery.toLowerCase()) || 
        (entry.id || '').toLowerCase().includes(searchQuery.toLowerCase()) || 
        (entry.score || 0).toString().includes(searchQuery)
      );

      const displayScores = filteredScores.slice(0, 50);

      const currentUserRecord = lastSavedEmpId ? scores.find(s => String(s.id).trim() === String(lastSavedEmpId).trim()) : null;
      const currentUserRank = currentUserRecord ? scores.findIndex(s => s.id === currentUserRecord.id) + 1 : 0;
      const isCurrentUserVisible = currentUserRecord ? displayScores.some(s => s.id === currentUserRecord.id) : false;
      
      return (
        <div className={`relative z-10 w-full max-w-lg p-4 sm:p-6 transition-all duration-700 transform flex flex-col h-[100dvh] justify-center ${isVisible ? 'translate-y-0 opacity-100 scale-100' : 'translate-y-12 opacity-0 scale-95'}`}>
          
          <div className={`relative overflow-hidden rounded-3xl border border-white/20 flex flex-col shadow-xl max-h-[90dvh] bg-black/80`} style={{ backfaceVisibility: 'hidden' }}>
            
            <div className="p-6 pb-4 border-b border-white/10 shrink-0">
              <div className="flex items-center justify-center gap-3 mb-6">
                <IconTrophy className="w-8 h-8 text-yellow-400" />

              </div>
              <div className="relative w-full">

            <div className="flex-1 overflow-y-auto custom-scrollbar p-6 pt-2">
              <div className="flex flex-col gap-3">
